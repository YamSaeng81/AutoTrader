from pathlib import Path
from urllib.parse import unquote
from collections import Counter
import hashlib
import json
import re
import subprocess

root = Path(__file__).resolve().parents[2]
docs = root / 'docs'
ev = docs / 'evidence/document-audit-20261112'
sha = lambda b: hashlib.sha256(b).hexdigest()
changes = json.loads((ev / 'changes.json').read_text(encoding='utf-8'))
checks = []
def check(name, ok):
    checks.append({'name': name, 'passed': bool(ok)})
    assert ok, name

for c in changes:
    p = root / c['path']
    before = root / c['archive']
    check('preserved raw bytes ' + c['path'], sha(before.read_bytes()) == c['before_sha256'])
    c['after_lines'] = len(p.read_text(encoding='utf-8').splitlines())
    c['after_bytes'] = len(p.read_bytes())
    c['after_sha256'] = sha(p.read_bytes())
(ev / 'changes.json').write_text(json.dumps(changes, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')

protected = json.loads((ev / 'protected-documents.json').read_text(encoding='utf-8'))
for rel, h in protected.items():
    check('fixed contract/reference unchanged ' + rel, sha((root / rel).read_bytes()) == h)

old_ops = (docs / 'OPS_DAILY_CHECK_ARCHIVE_20261112.md').read_text(encoding='utf-8')
new_ops = (docs / 'OPS_DAILY_CHECK.md').read_text(encoding='utf-8')
def commands(s):
    return re.findall(r'```bash\n(.*?)\n```', s, re.S)
check('OPS bash7 identical', len(commands(new_ops)) == 7 and commands(new_ops) == commands(old_ops))
body = lambda s: s[s.index('## 1. 무엇이 바뀌나'):s.index('## 6.')]
check('OPS entire sections1-5 identical', body(new_ops) == body(old_ops))
section = lambda s: s[s.index('### 3-1.'):s.index('## 4. 수동 실행')]
check('OPS extracted section3-1 identical', section(new_ops) == section(old_ops))

# Previous frozen review evidence and its original command bodies are not rewritten.
for folder in ['review-60777286', 'plan-reorganization-20261112']:
    base = docs / 'evidence' / folder
    for line in (base / 'SHA256SUMS').read_text(encoding='utf-8').splitlines():
        m = re.match(r'^([0-9a-f]{64})\s+\*?(.+)$', line)
        if m:
            p = base / m[2].removeprefix('./')
            check(f'prior evidence unchanged {folder}/{m[2]}', sha(p.read_bytes()) == m[1])

old = (docs / 'LIMITED_LIVE_TEST_PLAN_ARCHIVE_20261112.md').read_text(encoding='utf-8')
f = old[old.index('**F0. 비용 조건'):old.index('### 5-9. 11월 실운영 준비')]
check('F15 still identical', commands(f) == commands((docs / 'LIMITED_LIVE_TEST_RUNBOOK.md').read_text(encoding='utf-8')))

written = [root / c['path'] for c in changes] + [root / p for p in ['README.md','AGENTS.md','CLAUDE.md','docs/START_HERE.md','docs/CURRENT_HANDOFF.md','docs/evidence/document-audit-20261112/README.md']]
links = 0
for p in written:
    t = p.read_text(encoding='utf-8')
    check('no accidental Japanese text ' + p.relative_to(root).as_posix(), re.search('[\u3040-\u30ff]', t) is None)
    for dest in re.findall(r'\]\(([^)]+)\)', t):
        if re.match(r'^(https?|mailto):', dest) or dest.startswith('#'):
            continue
        raw, _, anchor = dest.partition('#')
        target = p.parent / unquote(raw)
        check(f'local link {p.relative_to(root).as_posix()}: {dest}', target.exists())
        links += 1
        if anchor and target.suffix.lower() == '.md':
            titles = re.findall(r'^#{1,6}\s+(.+)$', target.read_text(encoding='utf-8'), re.M)
            slugs = [re.sub(r'[^\w\-\s]', '', s.lower()).replace(' ', '-') for s in titles]
            check(f'local heading {p.relative_to(root).as_posix()}: {dest}', unquote(anchor) in slugs)

check('Claude import points to short guide', '@docs/START_HERE.md' in (root / 'CLAUDE.md').read_text(encoding='utf-8'))
check('new development submission retained', '397차' in (docs / 'PROGRESS.md').read_text(encoding='utf-8') and '397차' in (docs / 'PROGRESS_ARCHIVE_20261112.md').read_text(encoding='utf-8'))
check('new submission remains unaccepted', '총괄 검토 전' in (docs / 'PROGRESS.md').read_text(encoding='utf-8'))

five = [c for c in changes if Path(c['path']).name in ['PROGRESS.md','LIVE_READINESS.md','NOVEMBER_LIVE_UPGRADE.md','OPS_DAILY_CHECK.md','OVERNIGHT_20261111_HANDOFF.md']]
before_bytes = sum(c['before_bytes'] for c in five)
after_bytes = sum(c['after_bytes'] for c in five)
inventory = json.loads((ev / 'inventory-before.json').read_text(encoding='utf-8'))
summary = {'checks': len(checks), 'passed': sum(x['passed'] for x in checks), 'failed': 0,
           'inventory_files': len(inventory), 'inventory_by_kind': dict(Counter(x['kind'] for x in inventory)),
           'five_management_docs': {'before_lines': sum(c['before_lines'] for c in five), 'after_lines': sum(c['after_lines'] for c in five),
                                    'before_bytes': before_bytes, 'after_bytes': after_bytes, 'byte_reduction_pct': round(100*(before_bytes-after_bytes)/before_bytes,2)},
           'local_links': links, 'preserved_archives': len(changes), 'fixed_contracts_references_unchanged': len(protected),
           'checks_detail': checks}
(ev / 'verification.json').write_text(json.dumps(summary, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
print(json.dumps({k:v for k,v in summary.items() if k != 'checks_detail'},ensure_ascii=False))
