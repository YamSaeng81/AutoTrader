#!/bin/bash
# 사용: abrun.sh <결과이름> <역할 main|seed-other|none> [다른DB] — 새 DB 생성 후 AbChainV4PgTest 1회. 역할 none 은 AB_CHAIN_ROLE 미지정.
N=$1; ROLE=$2; OTHER=$3
SP="C:/Users/SANGJO~1/AppData/Local/Temp/claude/e30ef13592d895eb/59c77fac-b44d-4a67-9735-dc51cf649a08/scratchpad"
W=/d/tmp/abv4-84fb03d6; O=/d/tmp/v153-runs/v157/$N; rm -rf $O; mkdir -p $O/out; cd $W
DRV="D:/Claude Code/projects/crypto-auto-trader/docs/evidence/ab-create-v4-20261111/chain_driver_v4.py"
H=$(git rev-parse HEAD)
DB=p157_${N//-/_}_$RANDOM
docker exec v152-pg psql -U trader -d postgres -qc "CREATE DATABASE $DB" || { echo "DB 생성 실패"; exit 9; }
# 생성 창(15분 칸의 6분 이후, 12분 전) 대기 — 구동기가 C+6 전이면 보내지 않는다
while :; do m=$((10#$(date -u +%M) % 15)); [ $m -ge 6 ] && [ $m -le 11 ] && break; python -I -c "import time; time.sleep(20)"; done
{ echo "start $(date '+%F %T') head=$H db=$DB role=$ROLE other=${OTHER:-없음}"; echo "driver=$DRV"; sha256sum "$DRV"; } > $O/time.txt
git status --porcelain | grep -v '^??' > $O/dirty.txt
rm -rf web-api/build/test-results/test
E="AB_CHAIN_PG_URL=jdbc:postgresql://127.0.0.1:15437/$DB AB_CHAIN_DRIVER=$DRV AB_CHAIN_CODE_SHA=$H AB_CHAIN_OUT=$O/out"
[ "$ROLE" != none ] && E="$E AB_CHAIN_ROLE=$ROLE"
[ -n "$OTHER" ] && E="$E AB_CHAIN_OTHER_DB=$OTHER"
echo "env: $E" >> $O/time.txt
env AB_CHAIN_PG_URL=jdbc:postgresql://127.0.0.1:15437/$DB AB_CHAIN_DRIVER="$DRV" AB_CHAIN_CODE_SHA=$H AB_CHAIN_OUT=$O/out \
  ${ROLE:+$( [ "$ROLE" != none ] && echo AB_CHAIN_ROLE=$ROLE )} ${OTHER:+AB_CHAIN_OTHER_DB=$OTHER} \
  ./gradlew :web-api:test --rerun --console=plain --tests '*.AbChainV4PgTest' > $O/gradle.log 2>&1; rc=$?
echo "rc=$rc end $(date '+%F %T')" >> $O/time.txt
mkdir -p $O/xml/web-api; cp web-api/build/test-results/test/*.xml $O/xml/web-api/ 2>/dev/null
python -X utf8 -I "$SP/junitsum2.py" $O/xml > $O/summary.txt 2>&1
docker exec v152-pg psql -U trader -d $DB -tAc "select count(*) from paper_trading.virtual_balance" > $O/db-session-count.txt 2>&1
echo "$DB" > $O/db.txt
cat $O/time.txt $O/summary.txt
