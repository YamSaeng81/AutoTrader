#!/bin/bash
# 사용: trun.sh <작업트리> <결과이름> <시험패턴...> — web-api 시험만, 결과 폴더에 로그·XML·요약·시각
W=$1; N=$2; shift 2
SP="C:/Users/SANGJO~1/AppData/Local/Temp/claude/e30ef13592d895eb/59c77fac-b44d-4a67-9735-dc51cf649a08/scratchpad"
O=/d/tmp/v153-runs/v157/$N; rm -rf $O; mkdir -p $O; cd $W
H=$(git rev-parse HEAD); D=$(git status --porcelain | grep -v '^??' | wc -l)
echo "start $(date '+%F %T') $H dirty_tracked=$D args: $*" > $O/time.txt
git diff HEAD > $O/worktree.diff
rm -rf web-api/build/test-results/test
A=""; for t in "$@"; do A="$A --tests $t"; done
./gradlew :web-api:test --rerun --console=plain $A > $O/gradle.log 2>&1; rc=$?
echo "rc=$rc end $(date '+%F %T')" >> $O/time.txt
mkdir -p $O/xml/web-api; cp web-api/build/test-results/test/*.xml $O/xml/web-api/ 2>/dev/null
python -X utf8 -I "$SP/junitsum2.py" $O/xml > $O/summary.txt 2>&1
cat $O/time.txt $O/summary.txt | tail -15
